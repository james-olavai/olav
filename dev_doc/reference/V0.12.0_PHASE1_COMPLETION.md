```python
# ❌ 之前 - query_database.py (35 行)
def main(params: dict) -> dict:

# ✅ 之后 - query_database.py (118 行，但含 Pydantic 模型)
class QueryDatabaseInput(BaseModel):
    sql: str = Field(..., description="SQL query", min_length=1)
    timeout: int = Field(default=30, ge=1, le=300)
    
    @validator('sql')
    def validate_sql_not_empty(cls, v):
        if not v.strip():
            raise ValueError("SQL command cannot be empty")
        return v

@tool
def query_database(sql: str, timeout: int = 30) -> dict:
    """Execute SQL query"""
    args = QueryDatabaseInput(sql=sql, timeout=timeout)
    return main({"sql": args.sql, "timeout": args.timeout})
```

## 📝 技术细节

### @tool 装饰器实现

```python
from langchain_core.tools import tool

@tool
def query_database(sql: str, timeout: int = 30) -> dict:
    """Execute intelligent SQL query with automatic schema discovery."""
    # 使用 Pydantic 验证参数
    params = {"sql": sql, "timeout": timeout}
    return main(params)  # 调用原始实现 (向后兼容)
```

### Pydantic 模型层级

```
InputModel (用户输入)
    ↓ (验证)
main(params) (原始业务逻辑)
    ↓ (处理)
OutputModel (统一格式)
    ↓ (序列化)
JSON 返回
```

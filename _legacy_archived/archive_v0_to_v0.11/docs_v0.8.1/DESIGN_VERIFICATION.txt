from pydantic import BaseModel, validator
from typing import Optional

class TopologyLink(BaseModel):
    """拓扑链接数据模型 - 核心验证在这里"""
    
    local_device: str
    remote_device: str
    local_port: Optional[str] = None
    remote_port: Optional[str] = None
    layer: str  # "L1" or "L3"
    protocol: str  # "CDP", "LLDP", "OSPF", "BGP"
    confidence: float = 0.95
    
    @validator('local_device', 'remote_device', pre=True)
    def validate_device_names(cls, v):
        """关键验证: 设备名必须在已知设备列表中"""
        
        known_devices = get_known_devices()  # ['R1', 'R2', 'R3', 'R4', 'SW1', 'SW2']
        
        # 规范化输入
        v = str(v).strip()
        
        # ❌ 规则1: 拒绝IP地址
        if is_ip_address(v):
            raise ValueError(
                f"Invalid device '{v}': IP address not allowed (use device name)"
            )
        
        # ❌ 规则2: 拒绝通用占位符
        placeholders = {'Neighbor', 'Unknown', 'Total', 'Switch', '%', 'N/A'}
        if v in placeholders:
            raise ValueError(
                f"Invalid device '{v}': placeholder/header not device name"
            )
        
        # ❌ 规则3: 拒绝未知设备
        if v not in known_devices:
            raise ValueError(
                f"Unknown device '{v}'. Known devices: {sorted(known_devices)}"
            )
        
        return v

# 这就是全部验证逻辑！简洁有效。

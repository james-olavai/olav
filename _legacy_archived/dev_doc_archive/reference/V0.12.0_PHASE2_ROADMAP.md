```python
import pytest
from olav_skills.network_query.tools.query_database import QueryDatabaseInput

def test_query_database_input_valid():
    """Valid input should pass"""
    input_model = QueryDatabaseInput(sql="SELECT * FROM devices")
    assert input_model.sql == "SELECT * FROM devices"
    assert input_model.timeout == 30

def test_query_database_input_invalid_empty_sql():
    """Empty SQL should fail"""
    with pytest.raises(ValueError):
        QueryDatabaseInput(sql="")

def test_query_database_input_invalid_timeout():
    """Invalid timeout should fail"""
    with pytest.raises(ValueError):
        QueryDatabaseInput(sql="SELECT *", timeout=400)  # > 300
```
